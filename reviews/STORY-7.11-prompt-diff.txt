+class _FailureInjector:
+    """Module-local state for the once-mode failure inject flag.
+
+    Encapsulated in a class so unit tests can monkeypatch / reset it
+    without touching env vars.
+    """
+
+    def __init__(self) -> None:
+        self._once_armed: bool = False
+        self._mode_seen: str | None = None
+
+    def _refresh(self) -> None:
+        mode = os.environ.get(_FAILURE_INJECT_ENV, "").strip().lower() or None
+        if mode != self._mode_seen:
+            self._mode_seen = mode
+            # Re-arm `once` whenever the env var transitions in.
+            self._once_armed = mode == "once"
+
+    def should_inject(self) -> bool:
+        """Return True if this call should return the malformed fixture."""
+        self._refresh()
+        if self._mode_seen == "always":
+            return True
+        if self._mode_seen == "once" and self._once_armed:
+            self._once_armed = False  # consume
+            return True
+        return False
+
